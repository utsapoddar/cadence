# Cadence v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a single-file static web app that plays evidence-backed therapeutic tones with per-frequency citations, validated delivery methods, and study-matched durations.

**Architecture:** One `index.html` (markup + inline `<style>`) loads three ES modules: `audio-engine.js` (Web Audio synthesis for AM, binaural, notched pink noise), `ui.js` (card rendering + controls), `tinnitus-matcher.js` (pitch-matching modal). Frequency metadata lives in `frequencies.json`. Python script (`scripts/verify_tones.py`) generates reference WAVs for manual audio A/B. Hosted on GitHub Pages.

**Tech Stack:** Vanilla JS (ES modules), Web Audio API, zero build step, zero runtime deps. Python 3 + `numpy` + `scipy` for offline reference generation.

**Testing note:** Per spec, no JS unit-test framework (too few units). Correctness verified by the Python reference script + manual cross-platform smoke tests. Each task has an explicit manual verification step.

## Task 3: Audio engine — AM modulation (40 Hz gamma)

**Files:**
- Create: `audio-engine.js`

- [ ] **Step 1: Write AM synthesis skeleton**

```js
// audio-engine.js
let ctx = null;
let active = null; // { nodes: [], endAt: number, spec }

function getCtx() {
  if (!ctx) ctx = new (window.AudioContext || window.webkitAudioContext)();
  if (ctx.state === 'suspended') ctx.resume();
  return ctx;
}

function buildAM({ carrierHz, modHz, modDepth }) {
  const c = getCtx();
  const carrier = c.createOscillator();
  carrier.frequency.value = carrierHz;
  const mod = c.createOscillator();
  mod.frequency.value = modHz;
  const modGain = c.createGain();
  modGain.gain.value = modDepth;           // mod amplitude
  const out = c.createGain();
  out.gain.value = 0;                      // start silent, fade in
  mod.connect(modGain).connect(out.gain);  // gain is modulated around baseline
  out.gain.value = 1 - modDepth;           // DC offset so gain never goes negative
  carrier.connect(out).connect(c.destination);
  carrier.start(); mod.start();
  return { nodes: [carrier, mod, modGain, out], outGain: out };
}

export function play(spec, durationSec) {
  stop();
  let graph;
  if (spec.modality === 'am') graph = buildAM(spec.synthesis);
  else throw new Error(`unknown modality: ${spec.modality}`);
  fadeIn(graph.outGain, 1.0);
  const c = getCtx();
  const endAt = c.currentTime + durationSec;
  scheduleFadeOut(graph.outGain, endAt - 5, endAt);
  active = { graph, endAt, spec, stopTimer: setTimeout(stop, durationSec * 1000) };
}

function fadeIn(gainNode, seconds) {
  const c = getCtx();
  const base = gainNode.gain.value;
  gainNode.gain.cancelScheduledValues(c.currentTime);
  gainNode.gain.setValueAtTime(0, c.currentTime);
  gainNode.gain.linearRampToValueAtTime(base, c.currentTime + seconds);
}

function scheduleFadeOut(gainNode, startAt, endAt) {
  gainNode.gain.setValueAtTime(gainNode.gain.value, startAt);
  gainNode.gain.linearRampToValueAtTime(0, endAt);
}

export function stop() {
  if (!active) return;
  clearTimeout(active.stopTimer);
  const c = getCtx();
  const g = active.graph.outGain.gain;
  g.cancelScheduledValues(c.currentTime);
  g.setValueAtTime(g.value, c.currentTime);
  g.linearRampToValueAtTime(0, c.currentTime + 1);
  const nodes = active.graph.nodes;
  setTimeout(() => nodes.forEach(n => { try { n.stop && n.stop(); n.disconnect(); } catch {} }), 1100);
  active = null;
}

export function isPlaying() { return active !== null; }
export function getElapsed() { return active ? getCtx().currentTime - (active.endAt - 0) : 0; }
```

**Note on the AM math:** the synthesis uses a DC-offset gain (`1 - modDepth`) with an oscillator adding `modDepth × sin(2π·modHz·t)` on top. This produces amplitude modulation of the 1 kHz carrier at 40 Hz, matching the MIT GENUS audio protocol's audibility approach.

```js
function buildBinaural({ baseHz, beatHz }) {
  const c = getCtx();
  const left = c.createOscillator();
  const right = c.createOscillator();
  left.frequency.value = baseHz;
  right.frequency.value = baseHz + beatHz;
  const merger = c.createChannelMerger(2);
  const lGain = c.createGain(); lGain.gain.value = 0.25;
  const rGain = c.createGain(); rGain.gain.value = 0.25;
  left.connect(lGain).connect(merger, 0, 0);
  right.connect(rGain).connect(merger, 0, 1);
  const out = c.createGain(); out.gain.value = 1;
  merger.connect(out).connect(c.destination);
  left.start(); right.start();
  return { nodes: [left, right, lGain, rGain, merger, out], outGain: out };
}
```

```js
// Paul Kellet's pink noise algorithm
function makePinkNoiseBuffer(seconds = 10) {
  const c = getCtx();
  const rate = c.sampleRate;
  const buf = c.createBuffer(2, rate * seconds, rate);
  for (let ch = 0; ch < 2; ch++) {
    const data = buf.getChannelData(ch);
    let b0=0,b1=0,b2=0,b3=0,b4=0,b5=0,b6=0;
    for (let i = 0; i < data.length; i++) {
      const white = Math.random() * 2 - 1;
      b0 = 0.99886*b0 + white*0.0555179;
      b1 = 0.99332*b1 + white*0.0750759;
      b2 = 0.96900*b2 + white*0.1538520;
      b3 = 0.86650*b3 + white*0.3104856;
      b4 = 0.55000*b4 + white*0.5329522;
      b5 = -0.7616*b5 - white*0.0168980;
      data[i] = (b0+b1+b2+b3+b4+b5+b6+white*0.5362) * 0.11;
      b6 = white * 0.115926;
    }
  }
  return buf;
}
```

- [ ] **Step 2: Add `buildNotch` + dispatch**

```js
function buildNotch({ centerHz, notchBandwidthOctaves }) {
  const c = getCtx();
  const src = c.createBufferSource();
  src.buffer = makePinkNoiseBuffer(10);
  src.loop = true;
  const notch = c.createBiquadFilter();
  notch.type = 'notch';
  notch.frequency.value = centerHz;
  // Q = fc / bandwidth; bandwidth in Hz for 1-octave = fc * (2^0.5 - 2^-0.5) ≈ fc * 1.414 — but BiquadFilter Q uses different convention.
  // For 1-octave bandwidth on a notch: Q ≈ sqrt(2) / (2^octaves - 1) * 2^(octaves/2)  →  for 1 octave ≈ 1.414.
  notch.Q.value = Math.sqrt(2) / (Math.pow(2, notchBandwidthOctaves) - 1) * Math.pow(2, notchBandwidthOctaves/2);
  const out = c.createGain(); out.gain.value = 0.5;
  src.connect(notch).connect(out).connect(c.destination);
  src.start();
  return { nodes: [src, notch, out], outGain: out };
}
```

```js
const KEY = 'cadence.tinnitusHz';
let probeCtx = null, probeOsc = null;

function startProbe(hz) {
  if (!probeCtx) probeCtx = new (window.AudioContext || window.webkitAudioContext)();
  stopProbe();
  probeOsc = probeCtx.createOscillator();
  const g = probeCtx.createGain(); g.gain.value = 0.05;
  probeOsc.frequency.value = hz;
  probeOsc.connect(g).connect(probeCtx.destination);
  probeOsc.start();
}
function stopProbe() {
  if (probeOsc) { try { probeOsc.stop(); probeOsc.disconnect(); } catch {} probeOsc = null; }
}

export function getSavedPitch() {
  const v = localStorage.getItem(KEY);
  return v ? Number(v) : null;
}

export function openMatcher() {
  const dlg = document.getElementById('tinnitus-modal');
  const slider = document.getElementById('tinnitus-slider');
  const readout = document.getElementById('tinnitus-freq');
  const save = document.getElementById('tinnitus-save');
  const cancel = document.getElementById('tinnitus-cancel');

  slider.value = getSavedPitch() || 4000;
  readout.textContent = `${slider.value} Hz`;
  startProbe(Number(slider.value));

  const onInput = () => {
    readout.textContent = `${slider.value} Hz`;
    startProbe(Number(slider.value));
  };
  slider.addEventListener('input', onInput);

  return new Promise(resolve => {
    const cleanup = () => {
      stopProbe();
      slider.removeEventListener('input', onInput);
      save.onclick = cancel.onclick = null;
      dlg.close();
    };
    save.onclick = () => {
      const hz = Number(slider.value);
      localStorage.setItem(KEY, hz);
      cleanup(); resolve(hz);
    };
    cancel.onclick = () => { cleanup(); resolve(null); };
    dlg.showModal();
  });
}
```

```js
const state = { data: [], activeId: null, timerInterval: null };

function fmtDose(sec) {
  if (sec == null) return '';
  if (sec >= 3600) return `${sec/3600} hr`;
  if (sec >= 60) return `${sec/60} min`;
  return `${sec} s`;
}

function renderCard(card) {
  const cites = card.citations.map(c =>
    `<a class="cite" target="_blank" rel="noopener" href="${c.url}">${c.title} — ${c.journal}, ${c.year}</a>`
  ).join('');
  const disclosure = card.tier === 'C'
    ? `<div class="disclosure">${card.notes}</div>` : '';
  const playBtn = card.tier === 'C'
    ? '' : `<button data-play="${card.id}">Play · ${fmtDose(card.doseSec)}</button>`;
  const infoClass = card.tier === 'C' ? ' info-only' : '';
  return `
    <section class="card${infoClass}" data-id="${card.id}">
      <h2>${card.label}<span class="tier ${card.tier}">Tier ${card.tier}</span></h2>
      <div class="dose">Protocol dose: ${fmtDose(card.doseSec) || '—'}</div>
      <p>${card.notes}</p>
      ${disclosure}
      ${cites}
      ${playBtn}
    </section>`;
}

async function onPlayClick(id) {
  if (state.activeId === id) { stop(); state.activeId = null; updateButtons(); return; }
  const card = state.data.find(d => d.id === id);
  if (!card) return;
  let spec = card;
  if (card.modality === 'notch') {
    let pitch = getSavedPitch();
    if (!pitch) pitch = await openMatcher();
    if (!pitch) return;
```

```python
"""Generate reference WAV files for each card's synthesis spec.
Manual A/B against the browser output to verify the JS synthesis is correct.

Usage:
    python3 -m venv .venv && source .venv/bin/activate
    pip install -r scripts/requirements.txt
    python3 scripts/verify_tones.py
"""
import json, os, sys
import numpy as np
from scipy.io import wavfile
from scipy.signal import iirnotch, lfilter

SR = 44100
OUT_DIR = os.path.join(os.path.dirname(__file__), 'reference_wavs')
os.makedirs(OUT_DIR, exist_ok=True)
DURATION = 10  # seconds per reference clip

def am(carrier_hz, mod_hz, mod_depth):
    t = np.arange(SR * DURATION) / SR
    env = (1 - mod_depth) + mod_depth * np.sin(2*np.pi*mod_hz*t)
    sig = env * np.sin(2*np.pi*carrier_hz*t)
    return np.stack([sig, sig], axis=-1)  # stereo

def binaural(base_hz, beat_hz):
    t = np.arange(SR * DURATION) / SR
    left  = np.sin(2*np.pi*base_hz*t)
    right = np.sin(2*np.pi*(base_hz + beat_hz)*t)
    return np.stack([left, right], axis=-1)

def pink_noise(n):
    # Voss-McCartney approximation
    ncols = 16
    arr = np.random.randn(n, ncols)
    arr = np.cumsum(arr, axis=1) / np.arange(1, ncols+1)
    return arr.mean(axis=1)

def notch(center_hz, octaves=1.0):
    n = SR * DURATION
    pink = np.stack([pink_noise(n), pink_noise(n)], axis=-1)
    Q = np.sqrt(2) / (2**octaves - 1) * 2**(octaves/2)
    b, a = iirnotch(center_hz / (SR/2), Q)
    out = np.stack([lfilter(b, a, pink[:,0]), lfilter(b, a, pink[:,1])], axis=-1)
    return out

def to_int16(sig):
    sig = sig / max(1e-9, np.max(np.abs(sig)))
    return (sig * 0.5 * 32767).astype(np.int16)

def main():
    here = os.path.dirname(__file__)
    data = json.load(open(os.path.join(here, '..', 'frequencies.json')))
    for card in data:
        mod = card['modality']
        if mod == 'am':
            s = card['synthesis']
            sig = am(s['carrierHz'], s['modHz'], s['modDepth'])
        elif mod == 'binaural':
            s = card['synthesis']
            sig = binaural(s['baseHz'], s['beatHz'])
        elif mod == 'notch':
            sig = notch(4000, card['synthesis']['notchBandwidthOctaves'])  # demo center
        else:
            continue
        path = os.path.join(OUT_DIR, f"{card['id']}.wav")
        wavfile.write(path, SR, to_int16(sig))
        print(f"wrote {path}")

if __name__ == '__main__':
    main()
```
